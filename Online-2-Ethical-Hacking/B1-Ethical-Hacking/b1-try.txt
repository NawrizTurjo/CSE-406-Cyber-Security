// ইউনিভার্সাল এনকোডার - C1, B1, A1 সব কাজ করবে
// ------------------------------------------------------------
function encodePayload(payload, type) {
    // ১. প্রথমে বেসিক URL এনকোডিং করি (এটা <, >, /, ", ইত্যাদি কে % স্টাইলে কনভার্ট করে)
    let encoded = encodeURIComponent(payload);
    
    // ২. JavaScript-এর encodeURIComponent ' (সিঙ্গেল কোট) কে এনকোড করে না।
    //    কিন্তু SQL-এর ভিতরে ' বসালে ইনজেকশন ভেঙ্গে যায়। তাই ম্যানুয়ালি %27 বানাই।
    encoded = encoded.replace(/'/g, '%27');
    
    // ৩. টাইপ চেক করি
    if (type === 'double') {
        // ডাবল এনকোড: আগের এনকোডেড স্ট্রিংটাকে আবার পুরোপুরি এনকোড করি।
        // ফলে % -> %25, : -> %3A, / -> %2F ইত্যাদি হয়ে যায়।
        let doubleEncoded = encodeURIComponent(encoded);
        return doubleEncoded;
    }
    
    // সিঙ্গেল এনকোড (ডিফল্ট)
    return encoded;
}
